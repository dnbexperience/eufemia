import { List } from '@dnb/eufemia/src'
import { Value } from '@dnb/eufemia/src/extensions/forms'
import { fish_medium } from '@dnb/eufemia/src/icons'

export const Title = ({ overline, subline, long }) => (
  <List.Cell.Title>
    {overline && (
      <List.Cell.Title.Overline>Overline</List.Cell.Title.Overline>
    )}
    <span>
      {long
        ? 'A long title that wraps onto more lines, so the space on both sides of it shows in where the lines break'
        : 'Title'}
    </span>
    {subline && <List.Cell.Title.Subline>Subline</List.Cell.Title.Subline>}
  </List.Cell.Title>
)

export const variants = {
  plain: {},
  overline: { overline: true },
  subline: { subline: true },
  both: { overline: true, subline: true },
  long: { long: true },
}
